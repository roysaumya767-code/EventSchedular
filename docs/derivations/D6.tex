\section{Hold Model Specification}
The hold model benchmarks priority queues under steady-state conditions:
\begin{enumerate}
    \item A queue is populated with $M$ initial live events.
    \item At each iteration, the minimum event $e$ is extracted ($\text{deleteMin}$).
    \item A random timestamp increment $\Delta t$ is drawn from a probability distribution.
    \item A new event is inserted with timestamp $t(e) + \Delta t$ and a new sequence number.
    \item The live queue capacity $M$ remains invariant throughout execution.
\end{enumerate}

\section{Predicted Cost Model Formulated Prior to Measurement}
Execution cost per hold operation $(\text{deleteMin} + \text{insert})$ is modeled as:
\begin{align}
C_{\text{heap}}(M) &= \alpha_{\text{heap}} \log_2(M) + \beta_{\text{heap}} + \text{CachePenalty}_{\text{heap}}(M) \\
C_{\text{avl}}(M)  &= \alpha_{\text{avl}} \log_2(M) + \beta_{\text{avl}} + \text{CachePenalty}_{\text{avl}}(M)
\end{align}
Under cancellation probability $c \in [0.0, 0.95]$:
\begin{align}
C_{\text{heap}}(M, c) &= (1 - c) C_{\text{heap}}(M) + c \left[ O(1) + \frac{1}{M}\Theta(M) \right] \\
C_{\text{avl}}(M, c)  &= (1 - c) C_{\text{avl}}(M) + c \left[ \alpha_{\text{del}} \log_2(M) + \text{RotationCost} \right]
\end{align}

\subsection{Cache and Memory Locality Predictions}
\begin{itemize}
    \item \textbf{Structure A (Binary Min-Heap):}
    Stored contiguously in flat memory. For $M = 1{,}000$, footprint is $1{,}000 \times 24\text{ B} = 24\text{ KB}$, residing fully in L1 cache ($48\text{ KB}$).
    \textit{Prediction:} Heap throughput will be $1.8\times$ to $2.5\times$ higher than AVL for $M \le 10{,}000$.
    \item \textbf{Structure B (AVL Tree):}
    Each node is individually allocated ($48\text{ bytes}$ with pointers). Traversals perform pointer dereferences across fragmented heap locations, causing cache misses.
    \item \textbf{Crossover Prediction at $M \ge 100{,}000$:}
    At $M = 100{,}000$, heap footprint is $2.4\text{ MB}$, exceeding per-core L2 cache ($1.25\text{ MB}$). Sift-down accesses disjoint cache lines ($2i+1, 2i+2$), inducing L3 cache misses. The throughput gap will narrow significantly.
\end{itemize}

\section{Analysis of Discrepancies and Crossover Points}
\begin{enumerate}
    \item \textbf{Cache Locality Advantage:} For $M \le 10{,}000$, Heap consistently achieves $\approx 2\times$ the throughput of AVL. Hardware cycle counters (\texttt{\_\_rdtsc}) confirm Heap requires only $150$--$350$ cycles vs $300$--$700$ for AVL.
    \item \textbf{Crossover at $M = 100{,}000$:} Under Uniform increments, the ratio drops from $2.10\times$ down to $1.13\times$ as sift-downs exceed L2 cache capacity, inducing L3 cache transfers.
    \item \textbf{Cancellation Sweep Crossover:} At $25\%$--$50\%$ cancellation rates, AVL slightly outperforms Heap ($H/A \approx 0.84\text{x}$--$0.94\text{x}$) because true deletion reclaims nodes immediately, whereas Heap holds dead elements up to $2\times$ live size before purging.
\end{enumerate}
