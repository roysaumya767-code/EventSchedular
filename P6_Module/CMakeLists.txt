cmake_minimum_required(VERSION 3.16)
project(P6Bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(p6_bench src/benchmark.cpp)
enable_testing()
add_test(NAME harness_tests COMMAND p6_bench --self-test)
add_executable(p6_fault_test tests/fault_detection.cpp)
add_test(NAME rejects_corruption COMMAND p6_fault_test)
add_executable(p6_team_test tests/team_tests.cpp)
add_test(NAME team_regressions COMMAND p6_team_test)
